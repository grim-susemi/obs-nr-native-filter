cmake_minimum_required(VERSION 3.28...3.30)
project(obs-nr-filter VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Task2 supports Windows x64 only")
endif()
include(cmake/nr/Libobs.cmake)
add_library(obs-nr-filter MODULE src/plugin-main.c src/obs_filter.cpp)
target_compile_features(obs-nr-filter PRIVATE cxx_std_20)
target_link_libraries(obs-nr-filter PRIVATE OBS::libobs)
set_target_properties(obs-nr-filter PROPERTIES PREFIX "" OUTPUT_NAME "obs-nr-filter")
add_executable(nr-qa-host tools/host/main.cpp)
target_compile_features(nr-qa-host PRIVATE cxx_std_20)
target_link_libraries(nr-qa-host PRIVATE OBS::libobs)
target_compile_definitions(nr-qa-host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
enable_testing()
include(cmake/nr/Runtime.cmake)
include(cmake/nr/Core.cmake)
include(cmake/nr/Interop.cmake)
include(cmake/nr/Guides.cmake)
target_link_libraries(obs-nr-filter PRIVATE nr_interop)
target_sources(nr-qa-host PRIVATE tools/host/nr_interop_scenario.cpp)
target_link_libraries(nr-qa-host PRIVATE nr_interop)
target_link_libraries(nr-qa-host PRIVATE nr_flow_scenario)

target_compile_options(obs-nr-filter PRIVATE /W4 /WX $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
target_compile_options(nr-qa-host PRIVATE /W4 /WX /permissive-)
add_executable(frame_contract_tests tools/tests/frame_contract_tests.cpp)
target_include_directories(frame_contract_tests PRIVATE src)
target_compile_features(frame_contract_tests PRIVATE cxx_std_20)
add_test(NAME frame_contract COMMAND frame_contract_tests)
add_executable(sampler_contract_tests tools/tests/sampler_contract_tests.cpp)
target_include_directories(sampler_contract_tests PRIVATE src)
target_compile_features(sampler_contract_tests PRIVATE cxx_std_20)
target_compile_options(sampler_contract_tests PRIVATE /W4 /WX /permissive-)
add_test(NAME sampler_contract COMMAND sampler_contract_tests)
add_test(NAME host_runner COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/host_runner_tests.ps1" -Work "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME host_runner_flow COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/host_runner_tests.ps1" -Work "${CMAKE_CURRENT_SOURCE_DIR}" -Task8Only)
install(TARGETS obs-nr-filter LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
install(DIRECTORY data/ DESTINATION data/obs-plugins/obs-nr-filter)
